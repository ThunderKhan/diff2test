cmake_minimum_required(VERSION 3.20)
project(diff2test VERSION 0.1.0 LANGUAGES CXX)

function(diff2test_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()

add_executable(diff2test diff2test.cpp)
target_compile_features(diff2test PRIVATE cxx_std_20)
diff2test_warnings(diff2test)

include(CTest)
if(BUILD_TESTING)
    add_executable(json_tests tests/json_tests.cpp)
    target_compile_features(json_tests PRIVATE cxx_std_20)
    diff2test_warnings(json_tests)
    add_test(NAME json_tests COMMAND json_tests)
endif()
