cmake_minimum_required(VERSION 3.20)
project(diff2test VERSION 0.1.0 LANGUAGES CXX)

function(diff2test_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()

add_executable(diff2test diff2test.cpp)
target_compile_features(diff2test PRIVATE cxx_std_20)
diff2test_warnings(diff2test)

include(CTest)
if(BUILD_TESTING)
    foreach(test_target IN ITEMS json_tests dep_tests path_tests ctest_tests cmake_tests)
        add_executable(${test_target} tests/${test_target}.cpp)
        target_compile_features(${test_target} PRIVATE cxx_std_20)
        diff2test_warnings(${test_target})
        add_test(NAME ${test_target} COMMAND ${test_target})
    endforeach()
endif()
